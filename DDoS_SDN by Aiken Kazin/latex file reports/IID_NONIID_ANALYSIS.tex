\section{Current Implementation Status}

\subsection{✅ Current Setup: IID (Independent and Identically Distributed)}

The current federated learning setup uses \textbf{IID data distribution}.

\subsubsection{How It Works:}
\begin{enumerate}
    \item \textbf{Data Partitioning:} Each worker gets 50\% of the training data (\texttt{data\_partition=0.5})
    \item \textbf{Sampling Method:} Random sampling with worker-specific seed
    \begin{lstlisting}[language=Python]
seed = hash(worker_id) % 2**32
indices = np.random.RandomState(seed).choice(len(X_train), size=n_samples, replace=False)
    \end{lstlisting}
    \item \textbf{Result:} Each worker gets a random subset that preserves the overall class distribution
\end{enumerate}

\subsubsection{Verification Results:}
\begin{itemize}
    \item \textbf{Original Dataset:} Class 0: 60.91\%, Class 1: 39.09\%
    \item \textbf{Worker 1:} Class 0: 60.78\%, Class 1: 39.22\% (diff: 0.14\%)
    \item \textbf{Worker 2:} Class 0: 60.81\%, Class 1: 39.19\% (diff: 0.10\%)
    \item \textbf{Average Distribution Difference:} 0.12\%
\end{itemize}

✅ \textbf{Confirmed IID:} Workers have nearly identical class distributions

\section{IID vs Non-IID Comparison}

\subsection{IID (Independent and Identically Distributed) ✅ Current}

\textbf{Characteristics:}
\begin{itemize}
    \item ✅ Each worker has similar class distribution
    \item ✅ Random sampling preserves overall statistics
    \item ✅ Ideal for baseline FL experiments
    \item ✅ Easier convergence
    \item ✅ Represents ideal FL scenario
\end{itemize}

\textbf{Use Cases:}
\begin{itemize}
    \item Baseline experiments
    \item Algorithm comparison
    \item Proof of concept
    \item When data is naturally distributed
\end{itemize}

\textbf{Current Implementation:}
\begin{lstlisting}[language=Python]
# Random sampling with worker-specific seed
indices = np.random.RandomState(seed=hash(worker_id) % 2**32).choice(
    len(X_train), size=n_samples, replace=False
)
\end{lstlisting}

\subsection{Non-IID (Non-Independent and Identically Distributed)}

\textbf{Characteristics:}
\begin{itemize}
    \item ⚠️ Workers have different class distributions
    \item ⚠️ More realistic for real-world scenarios
    \item ⚠️ Harder convergence
    \item ⚠️ Better tests FL robustness
\end{itemize}

\textbf{Common Non-IID Scenarios:}
\begin{enumerate}
    \item \textbf{Class Imbalance:} Worker 1 gets 80\% class 0, Worker 2 gets 80\% class 1
    \item \textbf{Data Size Imbalance:} Different amounts of data per worker
    \item \textbf{Feature Distribution:} Different feature distributions per worker
    \item \textbf{Temporal Distribution:} Different time periods per worker
\end{enumerate}

\textbf{Example Non-IID Implementation:}
\begin{lstlisting}[language=Python]
# Partition by class labels (non-IID)
if worker_id.endswith('1'):
    # Worker 1: 80% class 0, 20% class 1
    class_0_indices = np.where(y_train == 0)[0]
    class_1_indices = np.where(y_train == 1)[0]
    n_class_0 = int(len(class_0_indices) * 0.8)
    n_class_1 = int(len(class_1_indices) * 0.2)
    indices = np.concatenate([
        np.random.choice(class_0_indices, n_class_0, replace=False),
        np.random.choice(class_1_indices, n_class_1, replace=False)
    ])
else:
    # Worker 2: 20% class 0, 80% class 1
    # Similar but reversed
\end{lstlisting}
